{Intercambiar} intercambiar (x,y) = (y,x)
{Espejar1} espejar (Left x) = Right x
{Espejar2} espejar (Right x) = Left x
{AsociarI} asociarI (x,(y,z)) = ((x,y),z)
{AsociarD} asociarD ((x,y),z)) = (x,(y,z))
{Flip} flip f x y = f y x
{Curry} curry f x y = f (x,y)
{Uncurry} uncurry f (x,y) = f x y
{Sucesor} sucesor n = n + 1
{Length0} length [] = 0
{Length1} length (_ : xs) = 1 + length xs
{Suma0} suma [] = 0
{Suma1} suma (x : xs) = x + suma xs
{SumaZero} suma Zero m = m
{SumaSuc} suma (Suc n) m = Suc (suma n) m
{NotTrue} not True = False
{NotFalse} not False = True
{First} fst (x, _) = x
{Second} snd (_, y) = y
{Swap} swap (x, y) = (y, x)

{Map0} map f [] = []
{Map1} map f (x : xs) = f x : map f xs
{Concat0} [] ++ ys = ys
{Concat1} (x : xs) ++ ys = x : (xs ++ ys)
{Foldr} foldr f z (x : xs) = f x (foldr f z xs) 
{Id} id x = x 
{Composición} (g . f) x = g (f x)
{Const} const x y = x
