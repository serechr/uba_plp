foldr :: (a -> b -> b) -> b -> [a] -> b
foldr _ z []     = z
foldr f z (x:xs) = f x (foldr f z xs)

foldl :: (b -> a -> b) -> b -> [a] -> b
foldl _ z []     = z
foldl f z (x:xs) = foldl f (f z x) xs

foldr1 :: (a -> a -> a) -> [a] -> a
foldr1 _ [x]    = x
foldr1 f (x:xs) = f x (foldr1 f xs)

foldl1 :: (a -> a -> a) -> [a] -> a
foldl1 _ [x]    = x
foldl1 f (x:xs) = foldl f x xs

map :: (a -> b) -> [a] -> [b]
map _ []     = []
map f (x:xs) = f x : map f xs

zipWith :: (a -> b -> c) -> [a] -> [b] -> [c]
zipWith _ [] _ = []
zipWith _ _ [] = []
zipWith f (x:xs) (y:ys) = f x y : zipWith f xs ys

all :: (a -> Bool) -> [a] -> Bool
all _ []     = True
all p (x:xs) = p x && all p xs

any :: (a -> Bool) -> [a] -> Bool
any _ []     = False
any p (x:xs) = p x || any p xs

null :: [a] -> Bool
null [] = True
null _  = False

nub :: Eq a => [a] -> [a]
nub [] = []
nub (x:xs) = x : nub (filter (/= x) xs)

sort :: Ord a => [a] -> [a]
sort [] = []
sort (x:xs) = sort [y | y <- xs, y <= x] ++ [x] ++ sort [y | y <- xs, y > x]

sortBy :: (a -> a -> Ordering) -> [a] -> [a]
sortBy _ [] = []
sortBy cmp (x:xs) = sortBy cmp [y | y <- xs, cmp y x /= GT] ++ [x] ++ sortBy cmp [y | y <- xs, cmp y x == GT]

mod :: Integral a => a -> a -> a
mod x y = x - (x `div` y) * y

odd :: Integral a => a -> Bool
odd x = x `mod` 2 /= 0

even :: Integral a => a -> Bool
even x = x `mod` 2 == 0

(++) :: [a] -> [a] -> [a]
[] ++ ys = ys
(x:xs) ++ ys = x : (xs ++ ys)

head :: [a] -> a
head (x:_) = x

tail :: [a] -> [a]
tail (_:xs) = xs

init :: [a] -> [a]
init [_] = []
init (x:xs) = x : init xs

last :: [a] -> a
last [x] = x
last (_:xs) = last xs

length :: [a] -> Int
length [] = 0
length (_:xs) = 1 + length xs

replicate :: Int -> a -> [a]
replicate 0 _ = []
replicate n x = x : replicate (n-1) x

repeat :: a -> [a]
repeat x = x : repeat x

iterate :: (a -> a) -> a -> [a]
iterate f x = x : iterate f (f x)

filter :: (a -> Bool) -> [a] -> [a]
filter _ [] = []
filter p (x:xs) 
  | p x       = x : filter p xs
  | otherwise = filter p xs

take :: Int -> [a] -> [a]
take 0 _      = []
take _ []     = []
take n (x:xs) = x : take (n-1) xs

drop :: Int -> [a] -> [a]
drop 0 xs      = xs
drop _ []      = []
drop n (_:xs)  = drop (n-1) xs

elem :: Eq a => a -> [a] -> Bool
elem _ []     = False
elem y (x:xs) = (y == x) || elem y xs

find :: (a -> Bool) -> [a] -> Maybe a
find _ [] = Nothing
find p (x:xs)
  | p x       = Just x
  | otherwise = find p xs

isNothing :: Maybe a -> Bool
isNothing Nothing = True
isNothing _       = False

isJust :: Maybe a -> Bool
isJust (Just _) = True
isJust _        = False

fromJust :: Maybe a -> a
fromJust (Just x) = x
fromJust Nothing  = error "Maybe.fromJust: Nothing"

maybe :: b -> (a -> b) -> Maybe a -> b
maybe d _ Nothing  = d
maybe _ f (Just x) = f x

lookup :: Eq a => a -> [(a, b)] -> Maybe b
lookup _ [] = Nothing
lookup k ((x, y):xys)
  | k == x    = Just y
  | otherwise = lookup k xys

reverse :: [a] -> [a]
reverse []     = []
reverse (x:xs) = reverse xs ++ [x]

concat :: [[a]] -> [a]
concat [] = []
concat (xs:xss) = xs ++ concat xss

union :: Eq a => [a] -> [a] -> [a]
union xs [] = xs
union xs (y:ys)
  | y `elem` xs = union xs ys
  | otherwise   = union (xs ++ [y]) ys

intersect :: Eq a => [a] -> [a] -> [a]
intersect [] _ = []
intersect (x:xs) ys
  | x `elem` ys = x : intersect xs ys
  | otherwise   = intersect xs ys

(>>=) :: Monad m => m a -> (a -> m b) -> m b
(>>=) mx f = do
  x <- mx
  f x

span :: (a -> Bool) -> [a] -> ([a], [a])
span _ [] = ([], [])
span p xs@(x:xs')
  | p x       = let (ys, zs) = span p xs' in (x:ys, zs)
  | otherwise = ([], xs)

takeWhile :: (a -> Bool) -> [a] -> [a]
takeWhile _ [] = []
takeWhile p (x:xs)
  | p x       = x : takeWhile p xs
  | otherwise = []

dropWhile :: (a -> Bool) -> [a] -> [a]
dropWhile _ [] = []
dropWhile p xs@(x:xs')
  | p x       = dropWhile p xs'
  | otherwise = xs

concatMap :: (a -> [b]) -> [a] -> [b]
concatMap _ [] = []
concatMap f (x:xs) = f x ++ concatMap f xs

and :: [Bool] -> Bool
and []     = True
and (x:xs) = x && and xs

or :: [Bool] -> Bool
or []     = False
or (x:xs) = x || or xs

sum :: Num a => [a] -> a
sum []     = 0
sum (x:xs) = x + sum xs

max :: Ord a => a -> a -> a
max x y
  | x >= y    = x
  | otherwise = y

maximum :: Ord a => [a] -> a
maximum [x] = x
maximum (x:xs) = max x (maximum xs)

maximumBy :: (a -> a -> Ordering) -> [a] -> a
maximumBy _ [x] = x
maximumBy cmp (x:xs) =
  let maxTail = maximumBy cmp xs
  in if x `cmp` maxTail == GT then x else maxTail

min :: Ord a => a -> a -> a
min x y
  | x <= y    = x
  | otherwise = y

minimum :: Ord a => [a] -> a
minimum [x] = x
minimum (x:xs) = min x (minimum xs)

minimumBy :: (a -> a -> Ordering) -> [a] -> a
minimumBy _ [x] = x
minimumBy cmp (x:xs) =
  let minTail = minimumBy cmp xs
  in if x `cmp` minTail == LT then x else minTail

(==) :: Eq a => a -> a -> Bool
(==) x y = x `eq` y where eq a b = not (a /= b)

(/=) :: Eq a => a -> a -> Bool
(/=) x y = not (x == y)

compare :: Ord a => a -> a -> Ordering
compare x y
  | x < y     = LT
  | x > y     = GT
  | otherwise = EQ

comparing :: Ord a => (b -> a) -> b -> b -> Ordering
comparing p x y = compare (p x) (p y)

not :: Bool -> Bool
not True  = False
not False = True

ord :: Char -> Int
ord c = fromEnum c

chr :: Int -> Char
chr n = toEnum n
